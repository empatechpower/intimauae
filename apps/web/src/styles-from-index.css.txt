    (function () {
      // Age gate
      var gate = document.getElementById('ageGate');
      if (localStorage.getItem('intimauae_age_verified') === 'true') gate.classList.remove('show');
      document.getElementById('ageEnter').onclick = function () {
        localStorage.setItem('intimauae_age_verified', 'true');
        gate.classList.remove('show');
      };
      document.getElementById('ageCancel').onclick = function () {
        window.location.href = 'https://www.google.com';
      };

      // Mobile nav
      document.getElementById('menuBtn').onclick = function () {
        document.getElementById('mainNav').classList.toggle('open');
      };

      // Hero slider
      var track = document.getElementById('heroTrack');
      var slides = track.children.length;
      var index = 0;
      var dots = document.getElementById('heroDots');
      for (var i = 0; i < slides; i++) {
        var b = document.createElement('button');
        b.type = 'button';
        b.setAttribute('aria-label', 'Go to slide ' + (i + 1));
        if (i === 0) b.className = 'on';
        (function (n) { b.onclick = function () { go(n); }; })(i);
        dots.appendChild(b);
      }
      function go(n) {
        index = (n + slides) % slides;
        track.style.transform = 'translateX(' + (-index * 100) + '%)';
        Array.prototype.forEach.call(dots.children, function (d, i) {
          d.className = i === index ? 'on' : '';
        });
      }
      document.getElementById('heroPrev').onclick = function () { go(index - 1); };
      document.getElementById('heroNext').onclick = function () { go(index + 1); };
      setInterval(function () { go(index + 1); }, 5500);

      // Products
      function money(n) {
        if (typeof formatMoney === 'function') return formatMoney(n);
        return '$' + Number(n).toFixed(2);
      }
      function card(p) {
        var img = (p.images && p.images[0]) || '';
        if (img && img.charAt(0) !== '/' && img.indexOf('http') !== 0) img = '/' + img;
        var sale = p.compare_at_price && p.compare_at_price > p.price;
        return (
          '<a class="product-card" href="/products/' + p.handle + '">' +
            '<div class="thumb"><img src="' + img + '" alt="' + String(p.title).replace(/"/g, '&quot;') + '" loading="lazy"></div>' +
            '<div class="meta">' +
              '<div class="vendor">' + p.category + '</div>' +
              '<div class="title">' + p.title + '</div>' +
              '<div class="price">' +
                (sale ? '<s>' + money(p.compare_at_price) + '</s>' : '') +
                money(p.price) +
              '</div>' +
            '</div>' +
          '</a>'
        );
      }
      function fill(id, list) {
        var el = document.getElementById(id);
        if (!el) return;
        el.innerHTML = list.map(card).join('') || '<p style="color:#9aa6b2">No products found.</p>';
      }

      if (typeof INTIMAUAE_PRODUCTS === 'undefined') {
        console.error('Product catalog missing');
        return;
      }
      var all = INTIMAUAE_PRODUCTS;
      fill('grid-full', all.filter(function (p) { return p.category === 'Full Body Products'; }));
      fill('grid-partial', all.filter(function (p) { return p.category === 'Partial Body Products'; }));
      fill('grid-trunk', all.filter(function (p) { return p.category === 'Trunk'; }));
    })();
  </script>
